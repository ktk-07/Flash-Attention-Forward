cmake_minimum_required(VERSION 3.18)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 86)
endif()
project(part_c LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../part_a/cpu_reference ${CMAKE_CURRENT_BINARY_DIR}/shared_reference)
add_library(part_c_attention STATIC
  src/flash_attention_1.cu
  src/flash_attention_1_tensor_core.cu
  src/flash_attention_2_cuda_core.cu
  src/flash_attention_2_tensor_core.cu)
target_include_directories(part_c_attention PUBLIC include)
target_link_libraries(part_c_attention PUBLIC CUDA::cudart)
set_target_properties(part_c_attention PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
target_compile_options(part_c_attention PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
add_executable(part_c_runner tests/kernel_runner.cpp tests/test_support.cpp)
target_link_libraries(part_c_runner PRIVATE part_c_attention attention_reference)
target_compile_features(part_c_runner PRIVATE cxx_std_17)
target_compile_definitions(part_c_runner PRIVATE PART_C_CUDA_COMPILER_VERSION="${CMAKE_CUDA_COMPILER_VERSION}")
include(CTest)
if(BUILD_TESTING)
  add_test(NAME part_c_correctness COMMAND part_c_runner --suite correctness --output ${CMAKE_CURRENT_BINARY_DIR}/ctest-results)
endif()
add_executable(part_c_evaluation tests/evaluation_runner.cpp tests/test_support.cpp)
target_link_libraries(part_c_evaluation PRIVATE part_c_attention attention_reference)
target_compile_features(part_c_evaluation PRIVATE cxx_std_17)
add_executable(part_c_input_contract tests/input_contract_test.cpp tests/test_support.cpp)
target_link_libraries(part_c_input_contract PRIVATE part_c_attention attention_reference)
target_compile_features(part_c_input_contract PRIVATE cxx_std_17)
if(BUILD_TESTING)
  add_test(NAME part_c_input_contract COMMAND part_c_input_contract)
endif()

# Dedicated Algorithm-1 invariants, including dense mode and per-forward reset.
add_executable(part_c_fa1_test tests/flash_attention_1_test.cpp tests/test_support.cpp)
target_link_libraries(part_c_fa1_test PRIVATE part_c_attention attention_reference)
target_compile_features(part_c_fa1_test PRIVATE cxx_std_17)
if(BUILD_TESTING)
  add_test(NAME part_c_fa1_invariants COMMAND part_c_fa1_test --output ${CMAKE_CURRENT_BINARY_DIR}/fa1-ctest-results)
endif()

if(BUILD_TESTING)
  add_test(NAME part_c_fa1_tensorcore_invariants COMMAND part_c_fa1_test --tensorcore --output ${CMAKE_CURRENT_BINARY_DIR}/fa1-tensorcore-ctest-results)
endif()
