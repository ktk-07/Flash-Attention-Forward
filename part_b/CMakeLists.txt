cmake_minimum_required(VERSION 3.18)
project(part_b LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../part_a/cpu_reference ${CMAKE_CURRENT_BINARY_DIR}/shared_reference)
add_library(dense_attention STATIC src/dense_flash_attention.cu)
target_include_directories(dense_attention PUBLIC include ../part_a/common)
target_link_libraries(dense_attention PUBLIC CUDA::cudart)
target_compile_features(dense_attention PUBLIC cxx_std_17 cuda_std_17)
target_compile_definitions(dense_attention PUBLIC NOMINMAX)
target_compile_options(dense_attention PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
add_executable(part_b_runner tests/evaluation_runner.cpp)
add_library(part_b_materialised_check STATIC ../part_a/gpu_baseline/qk.cu ../part_a/gpu_baseline/softmax.cu ../part_a/gpu_baseline/pv.cu)
target_include_directories(part_b_materialised_check PRIVATE ../part_a/common)
target_compile_features(part_b_materialised_check PRIVATE cuda_std_17)
target_compile_definitions(part_b_materialised_check PRIVATE NOMINMAX)
target_link_libraries(part_b_runner PRIVATE dense_attention attention_reference CUDA::cudart)
target_link_libraries(part_b_runner PRIVATE part_b_materialised_check)
enable_testing()
add_test(NAME shared_reference COMMAND attention_reference_test)
