cmake_minimum_required(VERSION 3.16)
project(part_a_cpu_reference LANGUAGES CXX)
enable_testing()
add_library(attention_reference src/attention_reference.cpp)
target_compile_features(attention_reference PUBLIC cxx_std_17)
set_target_properties(attention_reference PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(attention_reference PUBLIC include ${CMAKE_CURRENT_SOURCE_DIR}/../common)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(attention_reference PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(attention_demo examples/demo.cpp)
target_link_libraries(attention_demo PRIVATE attention_reference)
add_executable(reference_tests tests/reference_tests.cpp)
target_link_libraries(reference_tests PRIVATE attention_reference)
foreach(case tiny singleton uniform stability isolation causal_prefix irregular validation)
  add_test(NAME ${case} COMMAND reference_tests ${case})
endforeach()
